-- ========================================================= -- منصة العراق التعليمية -- قاعدة بيانات الطلاب والتفعيل والدخول -- ========================================================= create table if not exists public.codes ( id bigint generated by default as identity primary key, code text unique not null, type text not null default 'monthly', active boolean not null default true, used boolean not null default false, created_at timestamptz not null default now(), expires_at timestamptz ); create table if not exists public.attendance ( id bigint generated by default as identity primary key, student_name text not null default '', governorate text not null default '', school text default '', device_id text, attended_at timestamptz not null default now() ); alter table public.attendance add column if not exists user_id uuid; alter table public.attendance add column if not exists student_code text; alter table public.attendance add column if not exists grade text; alter table public.attendance add column if not exists activation_code text; alter table public.attendance add column if not exists activation_type text; alter table public.attendance add column if not exists activation_expires_at timestamptz; alter table public.attendance add column if not exists login_count integer not null default 0; alter table public.attendance add column if not exists last_login_at timestamptz; create unique index if not exists attendance_user_id_unique on public.attendance(user_id) where user_id is not null; create table if not exists public.activations ( id bigint generated by default as identity primary key, user_id uuid, student_code text, student_name text, code text not null, code_type text, device_id text, activated_at timestamptz not null default now(), expires_at timestamptz ); alter table public.activations add column if not exists user_id uuid; alter table public.activations add column if not exists student_code text; -- ========================================================= -- إنشاء رقم طالب تلقائي -- ========================================================= create or replace function public.generate_student_code() returns text language plpgsql security definer set search_path = public as $$ declare v_code text; begin loop v_code := 'STU-' || to_char(now(),'YYYY') || '-' || upper(substr(md5(random()::text || clock_timestamp()::text),1,6)); exit when not exists ( select 1 from public.attendance where student_code = v_code ); end loop; return v_code; end; $$; -- ========================================================= -- تسجيل الطالب -- ========================================================= create or replace function public.register_student( p_student_name text, p_governorate text, p_school text, p_grade text, p_device_id text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user uuid; v_student_code text; v_result jsonb; begin v_user := auth.uid(); if v_user is null then raise exception 'NOT_AUTHENTICATED'; end if; select student_code into v_student_code from public.attendance where user_id = v_user limit 1; if v_student_code is null then v_student_code := public.generate_student_code(); end if; insert into public.attendance ( user_id, student_code, student_name, governorate, school, grade, device_id, attended_at, last_login_at, login_count ) values ( v_user, v_student_code, trim(p_student_name), trim(p_governorate), trim(p_school), trim(p_grade), p_device_id, now(), now(), 1 ) on conflict (user_id) do update set student_name = excluded.student_name, governorate = excluded.governorate, school = excluded.school, grade = excluded.grade, device_id = excluded.device_id, attended_at = now(), last_login_at = now(); select jsonb_build_object( 'student_code', student_code, 'student_name', student_name, 'governorate', governorate, 'school', school, 'grade', grade, 'login_count', login_count, 'last_login_at', last_login_at ) into v_result from public.attendance where user_id = v_user limit 1; return v_result; end; $$; -- ========================================================= -- جلب بيانات الطالب الحالي -- ========================================================= create or replace function public.get_my_student() returns jsonb language sql security definer set search_path = public as $$ select coalesce( ( select jsonb_build_object( 'student_code', student_code, 'student_name', student_name, 'governorate', governorate, 'school', school, 'grade', grade, 'login_count', login_count, 'last_login_at', last_login_at, 'activation_code', activation_code, 'activation_type', activation_type, 'activation_expires_at', activation_expires_at ) from public.attendance where user_id = auth.uid() limit 1 ), '{}'::jsonb ); $$; -- ========================================================= -- تسجيل دخول جديد -- ========================================================= create or replace function public.record_student_login( p_device_id text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user uuid; begin v_user := auth.uid(); if v_user is null then raise exception 'NOT_AUTHENTICATED'; end if; update public.attendance set last_login_at = now(), login_count = coalesce(login_count,0) + 1, device_id = p_device_id, attended_at = now() where user_id = v_user; return public.get_my_student(); end; $$; -- ========================================================= -- تفعيل كود -- ========================================================= create or replace function public.redeem_activation_code( p_code text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user uuid; v_code public.codes%rowtype; v_expiry timestamptz; v_student public.attendance%rowtype; begin v_user := auth.uid(); if v_user is null then raise exception 'NOT_AUTHENTICATED'; end if; select * into v_code from public.codes where upper(trim(code)) = upper(trim(p_code)) and active = true and used = false for update; if not found then raise exception 'INVALID_CODE'; end if; if v_code.expires_at is not null and v_code.expires_at <= now() then raise exception 'CODE_EXPIRED'; end if; select * into v_student from public.attendance where user_id = v_user limit 1; if v_code.type = 'annual' then v_expiry := now() + interval '1 year'; else v_expiry := now() + interval '1 month'; end if; update public.codes set used = true, active = false where id = v_code.id; insert into public.activations ( user_id, student_code, student_name, code, code_type, expires_at ) values ( v_user, v_student.student_code, v_student.student_name, v_code.code, v_code.type, v_expiry ); update public.attendance set activation_code = v_code.code, activation_type = v_code.type, activation_expires_at = v_expiry where user_id = v_user; return jsonb_build_object( 'ok', true, 'code', v_code.code, 'type', v_code.type, 'expires_at', v_expiry ); end; $$; -- ========================================================= -- التحقق من اشتراك الطالب -- ========================================================= create or replace function public.get_my_access() returns jsonb language plpgsql security definer set search_path = public as $$ declare v_row public.attendance%rowtype; begin select * into v_row from public.attendance where user_id = auth.uid() limit 1; if not found then return jsonb_build_object( 'authorized', false ); end if; if v_row.activation_expires_at is null then return jsonb_build_object( 'authorized', false ); end if; if v_row.activation_expires_at <= now() then return jsonb_build_object( 'authorized', false, 'expired', true ); end if; return jsonb_build_object( 'authorized', true, 'type', v_row.activation_type, 'code', v_row.activation_code, 'expires_at', v_row.activation_expires_at ); end; $$; -- ========================================================= -- تجربة مجانية لمدة 30 يوم -- ========================================================= create or replace function public.start_free_trial() returns jsonb language plpgsql security definer set search_path = public as $$ declare v_user uuid; v_student public.attendance%rowtype; v_expiry timestamptz; begin v_user := auth.uid(); if v_user is null then raise exception 'NOT_AUTHENTICATED'; end if; select * into v_student from public.attendance where user_id = v_user limit 1; if not found then raise exception 'STUDENT_NOT_REGISTERED'; end if; if v_student.activation_expires_at is not null and v_student.activation_expires_at > now() then return jsonb_build_object( 'ok', true, 'existing', true, 'expires_at', v_student.activation_expires_at ); end if; v_expiry := now() + interval '30 days'; insert into public.activations ( user_id, student_code, student_name, code, code_type, expires_at ) values ( v_user, v_student.student_code, v_student.student_name, 'FREE-' || upper(substr(md5(random()::text),1,8)), 'trial', v_expiry ); update public.attendance set activation_code = 'FREE-TRIAL', activation_type = 'trial', activation_expires_at = v_expiry where user_id = v_user; return jsonb_build_object( 'ok', true, 'type', 'trial', 'expires_at', v_expiry ); end; $$; -- ========================================================= -- توليد كود من لوحة الإدارة -- ========================================================= create or replace function public.create_activation_code( p_admin_code text, p_type text ) returns text language plpgsql security definer set search_path = public as $$ declare v_code text; begin if p_admin_code <> 'admin123' then raise exception 'ADMIN_NOT_ALLOWED'; end if; loop v_code := upper( substr( md5(random()::text || clock_timestamp()::text), 1, 10 ) ); exit when not exists ( select 1 from public.codes where code = v_code ); end loop; insert into public.codes ( code, type, active, used ) values ( v_code, case when p_type = 'annual' then 'annual' else 'monthly' end, true, false ); return v_code; end; $$; -- السماح باستدعاء الدوال grant execute on function public.register_student(text,text,text,text,text) to anon, authenticated; grant execute on function public.get_my_student() to anon, authenticated; grant execute on function public.record_student_login(text) to anon, authenticated; grant execute on function public.redeem_activation_code(text) to anon, authenticated; grant execute on function public.get_my_access() to anon, authenticated; grant execute on function public.start_free_trial() to anon, authenticated; grant execute on function public.create_activation_code(text,text) to anon, authenticated;
